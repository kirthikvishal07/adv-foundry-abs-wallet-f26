3. THE OPERATION IS BUILT AND SIGNED
------------------------------------

Source: script/SendPackedUserOp.s.sol

    function generatedSignedUserOperation(bytes memory callData,
        HelperConfig.NetworkConfig memory config)
        public view returns (PackedUserOperation memory)
    {
        uint256 nonce = vm.getNonce(config.account);
        PackedUserOperation memory unsignedUserOp =
            _generateUnsignedUserOperation(callData, config.account, nonce);

        bytes32 userOpHash =
            IEntryPoint(config.entryPoint).getUserOpHash(unsignedUserOp);
        bytes32 digest = userOpHash.toEthSignedMessageHash();
        (v, r, s) = vm.sign(ANVIL_DEFAULT_KEY, digest);
        unsignedUserOp.signature = abi.encodePacked(r, s, v);
        return unsignedUserOp;
    }

    function _generateUnsignedUserOperation(bytes memory callData,
        address sender, uint256 nonce)
        internal pure returns (PackedUserOperation memory)
    {
        uint128 verificationGasLimit = 16777216;
        uint128 callGasLimit = verificationGasLimit;
        uint128 maxPriorityFeePerGas = 256;
        uint128 maxFeePerGas = maxPriorityFeePerGas;

        return PackedUserOperation({
            sender: sender,
            nonce: nonce,
            initCode: hex"",
            callData: callData,
            accountGasLimits: bytes32(uint256(verificationGasLimit) << 128 | callGasLimit),
            preVerificationGas: verificationGasLimit,
            gasFees: bytes32(uint256(maxPriorityFeePerGas) << 128 | maxFeePerGas),
            paymasterAndData: hex"",
            signature: hex""
        });
    }

Important local-test detail: config.account is the Anvil default wallet, which
is also made the owner of MinimalAccount. Therefore the signature recovers to
MinimalAccount.owner().  There is no paymaster because paymasterAndData is
empty; MinimalAccount pays for the operation itself.

Source: lib/account-abstraction/contracts/interfaces/PackedUserOperation.sol

4. INTERFACE DECLARATION VS IMPLEMENTATION
------------------------------------------

Source: lib/account-abstraction/contracts/interfaces/IEntryPoint.sol

    function handleOps(
        PackedUserOperation[] calldata ops,
        address payable beneficiary
    ) external;

The implementation that actually runs is:
Source: lib/account-abstraction/contracts/core/EntryPoint.sol

    function handleOps(
        PackedUserOperation[] calldata ops,
        address payable beneficiary
    ) external virtual nonReentrant {
        uint256 opslen = ops.length;
        UserOpInfo[] memory opInfos = new UserOpInfo[](opslen);
        unchecked {
            _iterateValidationPhase(ops, opInfos, address(0), 0);

            uint256 collected = 0;
            emit BeforeExecution();

            for (uint256 i = 0; i < opslen; i++) {
                collected += _executeUserOp(i, ops[i], opInfos[i]);
            }

            _compensate(beneficiary, collected);
        }
    }

Thus this one-element ops array follows this high-level path:

    randomUser
      -> EntryPoint.handleOps([packedUserOp], randomUser-as-beneficiary)
      -> EntryPoint validation phase
      -> MinimalAccount.validateUserOp(...)
      -> EntryPoint execution phase
      -> MinimalAccount.execute(usdc, 0, mint-calldata)
      -> ERC20Mock.mint(minimalAccount, 1e18)
      -> EntryPoint compensates randomUser with collected ETH fees


5. VALIDATION: ENTRYPOINT CALLS MINIMALACCOUNT
----------------------------------------------

Source: lib/account-abstraction/contracts/core/EntryPoint.sol

    function _iterateValidationPhase(
        PackedUserOperation[] calldata ops,
        UserOpInfo[] memory opInfos,
        address expectedAggregator,
        uint256 opIndexOffset
    ) internal virtual returns (uint256 opsLen) {
        opsLen = ops.length;
        for (uint256 i = 0; i < opsLen; i++) {
            UserOpInfo memory opInfo = opInfos[opIndexOffset + i];
            (uint256 validationData, uint256 pmValidationData) =
                _validatePrepayment(opIndexOffset + i, ops[i], opInfo);
            _validateAccountAndPaymasterValidationData(
                opIndexOffset + i, validationData, pmValidationData,
                expectedAggregator
            );
        }
    }

    function _validatePrepayment(uint256 opIndex,
        PackedUserOperation calldata userOp, UserOpInfo memory outOpInfo)
        internal virtual returns (uint256 validationData,
                                  uint256 paymasterValidationData)
    {
        _copyUserOpToMemory(userOp, outOpInfo.mUserOp);
        outOpInfo.userOpHash = getUserOpHash(userOp);
        uint256 requiredPreFund = _getRequiredPrefund(outOpInfo.mUserOp);
        outOpInfo.prefund = requiredPreFund;
        validationData = _validateAccountPrepayment(
            opIndex, userOp, outOpInfo, requiredPreFund
        );
        require(_validateAndUpdateNonce(outOpInfo.mUserOp.sender,
                outOpInfo.mUserOp.nonce), "AA25 invalid account nonce");
        ...
    }

    function _getRequiredPrefund(MemoryUserOp memory mUserOp)
        internal virtual pure returns (uint256 requiredPrefund)
    {
        uint256 requiredGas = mUserOp.verificationGasLimit
            + mUserOp.callGasLimit
            + mUserOp.paymasterVerificationGasLimit
            + mUserOp.paymasterPostOpGasLimit
            + mUserOp.preVerificationGas;
        requiredPrefund = requiredGas * mUserOp.maxFeePerGas;
    }

    function _validateAccountPrepayment(uint256 opIndex,
        PackedUserOperation calldata op, UserOpInfo memory opInfo,
        uint256 requiredPrefund) internal virtual returns (uint256 validationData)
    {
        address sender = opInfo.mUserOp.sender;
        uint256 missingAccountFunds = 0;
        if (opInfo.mUserOp.paymaster == address(0)) {
            uint256 bal = balanceOf(sender);
            missingAccountFunds = bal > requiredPrefund
                ? 0 : requiredPrefund - bal;
        }
        validationData = _callValidateUserOp(
            opIndex, op, opInfo, missingAccountFunds
        );
        if (opInfo.mUserOp.paymaster == address(0)) {
            if (!_tryDecrementDeposit(sender, requiredPrefund)) {
                revert FailedOp(opIndex, "AA21 didn't pay prefund");
            }
        }
    }

    function _callValidateUserOp(uint256 opIndex,
        PackedUserOperation calldata op, UserOpInfo memory opInfo,
        uint256 missingAccountFunds)
        internal virtual returns (uint256 validationData)
    {
        bytes memory callData = abi.encodeCall(IAccount.validateUserOp,
            (op, opInfo.userOpHash, missingAccountFunds));
        success := call(gasLimit, sender, 0, ...); // low-level call
        ...
    }

    function validateUserOp(PackedUserOperation calldata userOp,
        bytes32 userOpHash, uint256 missingAccountFunds)
        external requiredFromEntryPoint returns (uint256 validationData)
    {
        validationData = _validateSignature(userOp, userOpHash);
        _payPrefund(missingAccountFunds);
    }

    function _validateSignature(PackedUserOperation calldata userOp,
        bytes32 userOpHash) internal view returns (uint256 validationData)
    {
        bytes32 ethSignedMessageHash =
            MessageHashUtils.toEthSignedMessageHash(userOpHash);
        address signer = ECDSA.recover(ethSignedMessageHash, userOp.signature);
        if (signer != owner()) return SIG_VALIDATION_FAILED;
        return SIG_VALIDATION_SUCCESS; // 0
    }

    function _payPrefund(uint256 missingAccountFunds) internal {
        if (missingAccountFunds != 0) {
            (bool success,) = payable(msg.sender).call{
                value: missingAccountFunds, gas: type(uint256).max
            }("");
            (success);
        }
    }

Here msg.sender is EntryPoint, so _payPrefund sends the missing ETH from
MinimalAccount to EntryPoint. EntryPoint's receive() function records that ETH
as MinimalAccount's EntryPoint deposit:

Source: lib/account-abstraction/contracts/core/StakeManager.sol

    receive() external payable { depositTo(msg.sender); }

    function depositTo(address account) public virtual payable {
        uint256 newDeposit = _incrementDeposit(account, msg.value);
        emit Deposited(account, newDeposit);
    }

    function _tryDecrementDeposit(address account, uint256 amount)
        internal virtual returns (bool)
    {
        if (deposits[account].deposit < amount) return false;
        deposits[account].deposit -= amount;
        return true;
    }

7. ACTUAL GAS CHARGE, REFUND, AND BENEFICIARY PAYMENT
------------------------------------------------------

Source: lib/account-abstraction/contracts/core/EntryPoint.sol

    function _postExecution(IPaymaster.PostOpMode mode,
        UserOpInfo memory opInfo, bytes memory context, uint256 actualGas)
        internal virtual returns (uint256 actualGasCost)
    {
        address refundAddress = mUserOp.sender; // no paymaster in this test
        ...
        actualGasCost = actualGas * gasPrice;
        uint256 prefund = opInfo.prefund;
        if (prefund < actualGasCost) { ... }
        else {
            uint256 refund = prefund - actualGasCost;
            _incrementDeposit(refundAddress, refund);
            _emitUserOperationEvent(opInfo, success,
                                    actualGasCost, actualGas);
        }
    }

The prefund was reserved/debited before execution. Only actualGasCost is
charged. The unused part is NOT sent back to the account's wallet directly:
it is restored to MinimalAccount's EntryPoint deposit for future operations.

Source: lib/account-abstraction/contracts/core/EntryPoint.sol

    function _compensate(address payable beneficiary, uint256 amount)
        internal virtual
    {
        require(beneficiary != address(0), InvalidBeneficiary(beneficiary));
        currentUserOpHash = bytes32(0);
        (bool success, bytes memory ret) = beneficiary.call{value: amount}("");
        require(success, FailedSendToBeneficiary(beneficiary, amount, ret));
    }

In this test beneficiary is payable(randomUser), so the final fee amount is
sent as native ETH to randomUser. payable(randomUser) is ONLY an address type
conversion so the function can send native ETH to it; it sends no ETH by
itself. The transfer occurs later in beneficiary.call{value: amount}("").


COMPLETE CALLER / CALLEE MAP
============================

1. Foundry vm.prank makes randomUser the caller for one call.
2. randomUser -> EntryPoint.handleOps(ops, payable(randomUser)).
3. EntryPoint -> MinimalAccount.validateUserOp(packedUserOp, hash, missingFunds).
4. MinimalAccount -> EntryPoint (native ETH prefund, only if funds are missing).
5. EntryPoint -> MinimalAccount.execute(usdc, 0, mintCallData).
6. MinimalAccount -> ERC20Mock.mint(minimalAccount, AMOUNT).
7. EntryPoint -> randomUser (actual collected native-ETH fees).
8. The test reads ERC20Mock.balanceOf(minimalAccount) and sees AMOUNT.
