2. STEP-BY-STEP TRACE CHRONOLOGY
================================================================================

Step 1: MinimalAccountTest.t.sol
  - vm.prank(randomUser) sets msg.sender = randomUser (simulating a Bundler).
  - randomUser calls EntryPoint.handleOps(ops, payable(randomUser)).

Step 2: EntryPoint.sol :: handleOps()
  - Receives PackedUserOperation[] ops and address payable beneficiary.
  - Starts Phase 1: calls _iterateValidationPhase(ops, opInfos, address(0), 0).

Step 3: EntryPoint.sol :: _validatePrepayment() & _validateAccountPrepayment()
  - Computes userOpHash via getUserOpHash(userOp).
  - Calculates requiredPrefund = maxGas * maxFeePerGas.
  - Checks if MinimalAccount has deposited enough ETH in EntryPoint.
  - If not, sets missingAccountFunds = requiredPrefund - currentDeposit.
  - Calls _callValidateUserOp(opIndex, op, opInfo, missingAccountFunds).

Step 4: MinimalAccount.sol :: validateUserOp()
  - Verified by modifier: requiredFromEntryPoint (msg.sender == i_entryPoint).
  - Calls _validateSignature(userOp, userOpHash):
      * digest = MessageHashUtils.toEthSignedMessageHash(userOpHash)
      * signer = ECDSA.recover(digest, userOp.signature)
      * verifies signer == owner() -> returns SIG_VALIDATION_SUCCESS (0)
  - Calls _payPrefund(missingAccountFunds):
      * Transfers missing ETH from MinimalAccount's balance back to EntryPoint.
  - Returns validationData = 0 to EntryPoint.

Step 5: EntryPoint.sol :: _validateAccountAndPaymasterValidationData()
  - Confirms validationData == 0 (signature is valid and not expired/out of block range).
  - Validation Phase complete!

Step 6: EntryPoint.sol :: _executeUserOp()
  - Enters Phase 2: Execution Phase.
  - EntryPoint calls its own internal function: this.innerHandleOp(callData, opInfo, context).

Step 7: EntryPoint.sol :: innerHandleOp()
  - Performs: Exec.call(mUserOp.sender, 0, callData, callGasLimit).
  - Calls MinimalAccount.execute(dest, value, functionData).

Step 8: MinimalAccount.sol :: execute()
  - Verified by modifier: requiredFromEntryPointOrOwner (msg.sender == i_entryPoint).
  - Executes low-level call:
      (bool success, bytes memory result) = dest.call{value: value}(functionData);
  - Forwards callData to dest (ERC20Mock USDC).

Step 9: ERC20Mock.sol :: mint()
  - USDC contract runs mint(address(minimalAccount), 1e18).
  - MinimalAccount's USDC balance increases by 1e18 tokens.
  - Call returns success = true.

Step 10: EntryPoint.sol :: _postExecution() & _compensate()
  - EntryPoint calculates actualGas = preGas - gasleft().
  - Computes actualGasCost = actualGas * gasPrice.
  - Refunds any excess prefund ETH back to MinimalAccount's deposit.
  - Calls _compensate(beneficiary, collected):
      beneficiary.call{value: actualGasCost}("")
  - randomUser (the Bundler) receives ETH reimbursement.

    /// @inheritdoc IEntryPoint
    function handleOps(
        PackedUserOperation[] calldata ops,
        address payable beneficiary
    ) external virtual nonReentrant {
        uint256 opslen = ops.length;
        UserOpInfo[] memory opInfos = new UserOpInfo[](opslen);
        unchecked {
            // PHASE 1: VALIDATION LOOP
            _iterateValidationPhase(ops, opInfos, address(0), 0);

            uint256 collected = 0;
            emit BeforeExecution();

            // PHASE 2: EXECUTION LOOP
            for (uint256 i = 0; i < opslen; i++) {
                collected += _executeUserOp(i, ops[i], opInfos[i]);
            }

            // PHASE 3: COMPENSATE BUNDLER
            _compensate(beneficiary, collected);
        }
    }

    function _validateSignature(PackedUserOperation calldata userOp, bytes32 userOpHash)
        internal view returns (uint256 validationData)
    {
        bytes32 ethSignedMessageHash = MessageHashUtils.toEthSignedMessageHash(userOpHash);
        address signer = ECDSA.recover(ethSignedMessageHash, userOp.signature);
        if (signer != owner()) {
            return SIG_VALIDATION_FAILED; // returns 1
        }
        return SIG_VALIDATION_SUCCESS;    // returns 0
    }

    function innerHandleOp(
        bytes memory callData,
        UserOpInfo memory opInfo,
        bytes calldata context
    ) external virtual returns (uint256 actualGasCost) {
        uint256 preGas = gasleft();
        require(msg.sender == address(this), InternalFunction());
        MemoryUserOp memory mUserOp = opInfo.mUserOp;
        uint256 callGasLimit = mUserOp.callGasLimit;

        IPaymaster.PostOpMode mode = IPaymaster.PostOpMode.opSucceeded;
        if (callData.length > 0) {
            // EXECUTE: EntryPoint calls MinimalAccount with userOp.callData
            bool success = Exec.call(mUserOp.sender, 0, callData, callGasLimit);
            if (!success) {
                mode = IPaymaster.PostOpMode.opReverted;
            }
        }
        unchecked {
            uint256 actualGas = preGas - gasleft() + opInfo.preOpGas;
            return _postExecution(mode, opInfo, context, actualGas);
        }
    }
